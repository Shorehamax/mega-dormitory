cmake_minimum_required(VERSION 3.24)
project(MegaDormitory VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MEGA_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Vulkan REQUIRED)
find_package(glfw3 3.3 REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(ZLIB REQUIRED)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
add_library(mega_world src/world.cpp src/economy.cpp src/society.cpp src/persistence.cpp src/vehicles.cpp src/routines.cpp src/customization.cpp src/movement.cpp src/construction.cpp src/medicine.cpp src/operations.cpp src/transport.cpp src/office.cpp src/combat.cpp src/governance.cpp src/repairs.cpp src/housing_work.cpp src/social.cpp src/administration.cpp src/companies.cpp src/emergencies.cpp src/browser_import.cpp src/personal_items.cpp src/airborne.cpp src/washing.cpp src/autonomy.cpp src/structures.cpp src/roads.cpp src/vehicle_services.cpp)
target_include_directories(mega_world PUBLIC include PRIVATE third_party)
target_link_libraries(mega_world PRIVATE PkgConfig::ZSTD ZLIB::ZLIB)
target_compile_options(mega_world PRIVATE -Wall -Wextra -Wpedantic)
if(MEGA_SANITIZE)
  target_compile_options(mega_world PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mega_world PUBLIC -fsanitize=address,undefined)
endif()
find_program(GLSLC glslc REQUIRED)
foreach(SHADER scene.vert scene.frag shadow.vert shadow.frag pressure.comp skin.comp)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders
    COMMAND ${GLSLC} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER} -o ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER})
  list(APPEND SPIRV ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv)
endforeach()
add_custom_target(mega_shaders DEPENDS ${SPIRV})
add_library(mega_assets src/assets.cpp src/characters.cpp)
target_include_directories(mega_assets PUBLIC include PRIVATE third_party)
add_executable(mega_dormitory src/main.cpp src/renderer.cpp src/gpu_skinning.cpp src/ui.cpp src/application.cpp)
target_include_directories(mega_dormitory PRIVATE third_party)
target_link_libraries(mega_dormitory PRIVATE mega_world mega_assets Vulkan::Vulkan glfw)
target_compile_definitions(mega_dormitory PRIVATE MEGA_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets" MEGA_SHADERS="${CMAKE_CURRENT_BINARY_DIR}/shaders" MEGA_PROJECT="${CMAKE_CURRENT_SOURCE_DIR}")
add_dependencies(mega_dormitory mega_shaders)
enable_testing()
add_executable(mega_tests tests/world_tests.cpp)
target_link_libraries(mega_tests PRIVATE mega_world)
add_test(NAME world_invariants COMMAND mega_tests)
add_executable(mega_asset_tests tests/asset_tests.cpp)
target_include_directories(mega_asset_tests PRIVATE third_party)
target_link_libraries(mega_asset_tests PRIVATE mega_assets)
target_compile_definitions(mega_asset_tests PRIVATE MEGA_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
add_test(NAME asset_import COMMAND mega_asset_tests)
if(MEGA_SANITIZE)
  target_compile_options(mega_assets PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mega_assets PUBLIC -fsanitize=address,undefined)
endif()
add_executable(mega_ui_tests tests/ui_tests.cpp src/ui.cpp)
target_include_directories(mega_ui_tests PRIVATE include third_party)
target_compile_definitions(mega_ui_tests PRIVATE MEGA_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_link_libraries(mega_ui_tests PRIVATE glfw)
if(MEGA_SANITIZE)
  target_compile_options(mega_ui_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mega_ui_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME multilingual_ui COMMAND mega_ui_tests)
add_executable(mega_import_tests tests/import_tests.cpp)
target_include_directories(mega_import_tests PRIVATE third_party)
target_link_libraries(mega_import_tests PRIVATE mega_world ZLIB::ZLIB)
add_test(NAME browser_migration_and_personal_items COMMAND mega_import_tests)

find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME localization_catalogues COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_locales.py)
endif()
option(MEGA_VULKAN_TESTS "Enable display-dependent Vulkan skinning integration checks" OFF)
if(MEGA_VULKAN_TESTS)
  add_test(NAME vulkan_skinning COMMAND mega_dormitory --skin-test --frames 1 --width 1000 --height 720)
  add_test(NAME vulkan_skinning_software COMMAND mega_dormitory --software --skin-test --frames 1 --width 1000 --height 720)
  add_test(NAME vulkan_autonomy_playthrough COMMAND mega_dormitory --playtest-autonomy --width 1000 --height 720)
  add_test(NAME vulkan_construction_playthrough COMMAND mega_dormitory --playtest-construction --width 1000 --height 720)
  add_test(NAME vulkan_vehicle_playthrough COMMAND mega_dormitory --playtest-vehicles --width 1000 --height 720)
  set_tests_properties(vulkan_vehicle_playthrough vulkan_construction_playthrough vulkan_skinning vulkan_skinning_software vulkan_autonomy_playthrough PROPERTIES
    ENVIRONMENT "VK_INSTANCE_LAYERS=VK_LAYER_KHRONOS_validation"
    FAIL_REGULAR_EXPRESSION "Validation Error;VUID-")
endif()

add_executable(mega_physics_tests tests/physics_tests.cpp)
target_link_libraries(mega_physics_tests PRIVATE mega_world)
add_test(NAME persistent_body_physics COMMAND mega_physics_tests)

add_executable(mega_washing_tests tests/washing_tests.cpp)
target_link_libraries(mega_washing_tests PRIVATE mega_world)
add_test(NAME communal_washing COMMAND mega_washing_tests)

add_executable(mega_autonomy_tests tests/autonomy_tests.cpp)
target_link_libraries(mega_autonomy_tests PRIVATE mega_world)
add_test(NAME resident_autonomy COMMAND mega_autonomy_tests)

add_executable(mega_structure_tests tests/structure_tests.cpp)
target_link_libraries(mega_structure_tests PRIVATE mega_world)
add_test(NAME flexible_construction COMMAND mega_structure_tests)

add_executable(mega_traffic_tests tests/traffic_tests.cpp)
target_link_libraries(mega_traffic_tests PRIVATE mega_world)
add_test(NAME vehicle_traffic COMMAND mega_traffic_tests)
