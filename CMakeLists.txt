find_program(GLSLC glslc REQUIRED)
foreach(SHADER scene.vert scene.frag shadow.vert shadow.frag pressure.comp skin.comp)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders
    COMMAND ${GLSLC} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER} -o ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER})
  list(APPEND SPIRV ${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv)
endforeach()
add_custom_target(mega_shaders DEPENDS ${SPIRV})

# A relocatable Linux package keeps runtime data beside the executable; saves live in user data.
file(GENERATE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/package/runtime-layout.json CONTENT "{\"layout\":1,\"name\":\"Mega Dormitory\",\"version\":\"${PROJECT_VERSION}\",\"channel\":\"development\"}\n")
install(TARGETS mega_dormitory RUNTIME DESTINATION .)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/package/runtime-layout.json DESTINATION .)
install(DIRECTORY third_party/ DESTINATION docs/third_party FILES_MATCHING PATTERN "*.LICENSE" PATTERN "*manifest.json" PATTERN "asset-decoders.json")
install(DIRECTORY assets DESTINATION . PATTERN "__pycache__" EXCLUDE)
install(FILES ${SPIRV} DESTINATION shaders)
install(FILES README.md DESTINATION .)
install(DIRECTORY docs DESTINATION .)
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME mega-dormitory)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Mega Dormitory native development build")
set(CPACK_PACKAGE_FILE_NAME "mega-dormitory-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
